How to Send Confirmation Emails From Your Website

Integration Guide

You can trigger confirmation emails from any website frontend or backend using our REST API or drop-in SDKs: 

<!-- 1. Include the drop-in client SDK in your website -->
<script src="http://localhost:3000/sdk/website-tracker.js"></script>

<script>
  // 2. Initialize with your current user
  WebsiteActivityTracker.init({
    endpoint: 'http://localhost:3000/api/activity/notify',
    user: {
      email: 'user@example.com',
      name: 'Alex Morgan'
    }
  });

  // 3. Trigger confirmation emails on any website action:
  
  // When user logs in:
  await WebsiteActivityTracker.trackLogin();

  // When user changes password:
  await WebsiteActivityTracker.trackPasswordChange();

  // When user updates their profile:
  await WebsiteActivityTracker.trackProfileUpdate(['Phone Number', 'Billing Address']);

  // When user places an order:
  await WebsiteActivityTracker.trackOrder('ORD-12345', '$89.00', [
    { name: 'Website Pro Plan', qty: 1, price: '$89.00' }
  ]);

  // Any custom activity:
  await WebsiteActivityTracker.track('file_export', {
    title: 'Financial Report Downloaded',
    description: 'Your export was successfully generated.',
    metadata: { 'File': 'q3_report.pdf', 'Size': '2.4MB' }
  });
</script>
// In your Node.js / Express website backend:
const { ActivityNotifier } = require('./client-sdk/node-client');

const notifier = new ActivityNotifier({
  endpoint: 'http://localhost:3000/api/activity/notify'
});

// Trigger confirmation email inside your route handler:
app.post('/api/user/login', async (req, res) => {
  // ... authenticate user ...
  
  // Send activity confirmation email asynchronously
  await notifier.notifyLogin({
    email: user.email,
    name: user.name
  }, req);

  res.json({ success: true });
});

// Trigger on password change:
await notifier.notifyPasswordChanged({ email: user.email, name: user.name }, req);

// Trigger on custom activity:
await notifier.notifyCustomActivity({ email: user.email, name: user.name }, {
  type: 'security_setting_updated',
  title: 'Two-Factor Authentication Enabled',
  description: 'Two-factor authentication has been enabled for your website account.',
  metadata: { 'Method': 'Authenticator App (TOTP)' }
});
# In your Python / Flask / Django / FastAPI website:
from client_sdk.python_client import ActivityNotifier

notifier = ActivityNotifier(endpoint="http://localhost:3000/api/activity/notify")

# Trigger on sign-in:
notifier.notify_login(
    user={"email": "alice@example.com", "name": "Alice"},
    ip=request.remote_addr,
    user_agent=request.headers.get("User-Agent")
)

# Trigger on password change:
notifier.notify_password_changed(
    user={"email": "alice@example.com", "name": "Alice"},
    ip=request.remote_addr
)

# Trigger on custom website activity:
notifier.notify_custom(
    user={"email": "alice@example.com", "name": "Alice"},
    title="Invoice Paid Successfully",
    description="Your monthly payment of $49.00 was processed.",
    metadata={"Invoice Number": "INV-2026-089", "Card": "Visa ending in 4242"},
    action_url="https://yourwebsite.com/billing"
)
curl -X POST http://localhost:3000/api/activity/notify \
  -H "Content-Type: application/json" \
  -d '{
    "user": {
      "email": "user@example.com",
      "name": "Alex Morgan"
    },
    "activity": {
      "type": "login",
      "title": "New Sign-In Detected",
      "ip": "203.0.113.1",
      "device": "Chrome on Windows 11",
      "actionUrl": "https://yourwebsite.com/security"
    }
  }'

Switching from Test Mode to Real SMTP:

To send real emails to inboxes via Gmail, SendGrid, Outlook, etc., edit the .env file: 

USE_ETHEREAL=false
SMTP_HOST=smtp.gmail.com
SMTP_PORT=587
SMTP_USER=your_email@gmail.com
SMTP_PASS=your_gmail_app_password
EMAIL_FROM="Your Website <no-reply@yourwebsite.com>"
